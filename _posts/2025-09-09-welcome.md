```python
def batting_average(hits, at_bats):
    """Return a batting average rounded to three decimal places."""
    if at_bats <= 0:
        raise ValueError("at_bats must be positive")
    return round(hits / at_bats, 3)


print(batting_average(150, 500))  # .300
```
